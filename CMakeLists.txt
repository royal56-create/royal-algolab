cmake_minimum_required(VERSION 3.16)
project(RoyalAlgoLab VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(royal_algolab src/main.cpp)
target_include_directories(royal_algolab PRIVATE include)
if(MSVC)
  target_compile_options(royal_algolab PRIVATE /W4 /permissive-)
else()
  target_compile_options(royal_algolab PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(algolab_tests tests/test_algolab.cpp)
  target_include_directories(algolab_tests PRIVATE include)
  add_test(NAME algolab_tests COMMAND algolab_tests)
endif()
